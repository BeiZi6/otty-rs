icon     = "theSVG:ssh"
detect   = ["ssh"]

# ssh_config(5) FILES: ~/.ssh/config is the per-user client config. The
# Include directive makes ~/.ssh/config.d a common split-config convention.
paths = ["~/.ssh/config", "~/.ssh/config.d/"]

# known_hosts is per-machine trust state that both ssh and the user rewrite
# constantly; syncing it produces conflicts with no benefit.
exclude = ["~/.ssh/known_hosts", "~/.ssh/known_hosts.old", "~/.ssh/authorized_keys"]

# Private keys. `~/.ssh/id_*` deliberately also catches the matching .pub
# files: over-classifying a public key costs a little convenience, and this
# list is a floor, not a ceiling.
secret = ["~/.ssh/id_*", "~/.ssh/*.pem", "~/.ssh/*.key"]
