        # A line that never BECAME a command reports 127, the code every POSIX
        # shell uses for it. PowerShell has no exit code of its own here -- both
        # a mistyped name and a build that failed arrive as 1 -- so a consumer
        # (the autocomplete history, which must not offer a typo back) cannot
        # tell them apart without this. FullyQualifiedErrorId carries the .NET
        # type name and is NOT localized, unlike the error text, and it is there
        # in every ErrorView including pwsh 7's ConciseView.
        if (-not $__otty_ok) {
          try {
            $base = if ($global:__OttyErrCount -is [int]) { $global:__OttyErrCount } else { [int]::MaxValue }
            if ($global:Error.Count -gt $base) {
              $er = $global:Error[0]
              if ($er -and ($er.FullyQualifiedErrorId -is [string]) -and
                  $er.FullyQualifiedErrorId.StartsWith('CommandNotFoundException')) { $code = 127 }
            }
          } catch {}
        }
        $out += "$e]133;D;$code$b"
        $global:__OttyCmdRan = $false
      }

      # nested in the __OttyCmdRan block above: the two flags are set together

      # so it does not ride on another flag staying in sync.
      if ($global:__OttyProgressActive) {
