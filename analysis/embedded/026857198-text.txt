__otty_should_wrap otty && function otty {
    if [[ "$1" == jump ]]; then
        shift
        __otty_jump "$@"

__otty_should_wrap learn && function learn {
    if [[ -n "$OTTY_FLAGS_DIR" && -f "$OTTY_FLAGS_DIR/omit_prefix" ]]; then
        command otty learn "$@"
    else
        command learn "$@"
    fi
}
