;; --- File writes (severely restricted) ---
;; Tools constantly redirect to /dev/null O_RDWR; denying writes there
;; breaks ssh and many others ("Couldn't open /dev/null").  /dev/stdout
;; and /dev/stderr are the captured pipes from our parent.
(allow file-write*
  (literal "/dev/null")
  (literal "/dev/stdout")
  (literal "/dev/stderr")
  (literal "/dev/tty"))
