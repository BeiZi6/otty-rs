— that one falls
            # through to a system `jump` whenever omit_prefix is off.
            case "$cmd" in
                jump)
                    functions[$name]='__otty_jump "$@"'
                    ;;
                *)
                    functions[$name]="command otty $cmd \"\$@\""
                    ;;
            esac
        done
    }
