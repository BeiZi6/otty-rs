                app  TEXT NOT NULL,

            );

                seq         INTEGER PRIMARY KEY AUTOINCREMENT,
                cmdline     TEXT NOT NULL,

                finished_ms INTEGER NOT NULL,

            );
            CREATE TABLE IF NOT EXISTS seen_binary (
                name    TEXT PRIMARY KEY,
