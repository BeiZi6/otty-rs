# Otty's remote shim: writes the request to the terminal as OSC 8801 for the
# Otty at the other end, and runs nothing locally. See src/shim.rs.

cmd=${1:-}
[ $# -gt 0 ] && shift

usage() {
    cat <<'EOF'
otty - hand a request to the Otty terminal on your own computer.

  otty view <path>...   open read-only
  otty edit <path>...   open in the editor
  otty jump [query]     jump to a recent directory
  otty learn <path>     remember a directory
  otty ignore <path>    stop remembering one

Anything else otty does needs the real binary, which this host has not got.
EOF
}

case "$cmd" in
    view|edit|jump|learn|ignore) ;;
    help|-h|--help) usage; exit 0 ;;
    '') usage >&2; exit 1 ;;
    *)
        printf 'otty %s is not available on a remote host: only view, edit, jump, learn and ignore travel over the terminal.\n' "$cmd" >&2
        exit 1 ;;
esac

    *[[:cntrl:]]*) printf 'otty: control characters cannot be sent.\n' >&2; exit 1 ;;
esac
