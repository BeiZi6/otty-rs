— $XDG_CONFIG_HOME/nvim on Unix,
# $LOCALAPPDATA/nvim on Windows. Everything under it is configuration except
# the caches plugin managers drop there.
paths = ["~/.config/nvim/", "%LOCALAPPDATA%/nvim/"]

# The first three repeat under %LOCALAPPDATA%: the Windows config tree grows
# the same plugin-manager leftovers, and an exclude that names only the Unix
# copy leaves them syncing from a Windows machine. The last three are the Unix
# data/state/cache dirs, which have no counterpart under the config path.
exclude = [
  "~/.config/nvim/.git/",
  "~/.config/nvim/spell/",
  "~/.config/nvim/plugin/packer_compiled.lua",
  "%LOCALAPPDATA%/nvim/.git/",
  "%LOCALAPPDATA%/nvim/spell/",
  "%LOCALAPPDATA%/nvim/plugin/packer_compiled.lua",
  "~/.local/share/nvim/",
  "~/.local/state/nvim/",
  "~/.cache/nvim/",
]

# show up (provider binaries, LSP roots), but they appear mid-expression

# heuristics.
